set(PROJ_NAME "apu")
set(PROJ_NAMESPACE "nes")

# Project Source
set(PROJ_SOURCE
	src/apu.cpp
	${CMAKE_SOURCE_DIR}/thirdparty/miniaudio/miniaudio.c
)

set(PROJ_HEADERS
)

set(PROJ_PUBLIC_HEADERS
	include/apu.h
	${CMAKE_SOURCE_DIR}/thirdparty/miniaudio/miniaudio.c
)

add_library(${PROJ_NAME} STATIC ${PROJ_SOURCE} ${PROJ_HEADERS} ${PROJ_PUBLIC_HEADERS})
add_library(${PROJ_NAMESPACE}::${PROJ_NAME} ALIAS ${PROJ_NAME})

target_include_directories(${PROJ_NAME}
	PUBLIC
		$<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/include>
		$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
	PRIVATE
		${CMAKE_SOURCE_DIR}/thirdparty/miniaudio
)

target_link_libraries(${PROJ_NAME}

)

# Install
install(TARGETS ${PROJ_NAME}
    EXPORT ${PROJ_NAME}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
    INCLUDES DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
	PUBLIC_HEADER
)
